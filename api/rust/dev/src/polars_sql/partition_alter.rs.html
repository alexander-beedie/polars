1//! `ALTER TABLE ... SET PARTITIONED BY (...)` and `ALTER TABLE ... RESET PARTITIONED BY`, which
2//! sqlparser does not parse yet.
3//!
4//! Temporary: remove once sqlparser parses these statements.
5
6use polars_core::prelude::*;
7use sqlparser::ast::{Expr, ObjectName, Statement};
8use sqlparser::keywords::Keyword;
9use sqlparser::parser::{Parser, ParserError};
10use sqlparser::tokenizer::Token;
11
12use crate::sql_expr::{sql_parser, to_sql_interface_err};
13
14/// A parsed `ALTER TABLE [IF EXISTS] <name> SET PARTITIONED BY (<fields>)` or
15/// `ALTER TABLE [IF EXISTS] <name> RESET PARTITIONED BY`.
16#[derive(Clone, Debug)]
17pub(crate) struct PartitionAlter {
18    pub(crate) name: ObjectName,
19    pub(crate) if_exists: bool,
20    /// The partition fields, as written; empty for `RESET PARTITIONED BY`.
21    pub(crate) fields: Vec<Expr>,
22}
23
24/// The statements of `query`, or its partition change.
25pub(crate) fn parse_statements_or_alter(
26    query: &str,
27) -> PolarsResult<Result<Vec<Statement>, PartitionAlter>> {
28    let (mut parser, versions) = sql_parser(query)?;
29    if let Some(alter) = parse_partition_alter(&mut parser)? {
30        // a time travel clause has no table to read here
31        versions.refuse()?;
32        return Ok(Err(alter));
33    }
34    let mut stmts = parser.parse_statements().map_err(to_sql_interface_err)?;
35    versions.attach(&mut stmts)?;
36    Ok(Ok(stmts))
37}
38
39/// The single statement of `query`, or its partition change.
40pub(crate) fn parse_single_statement_or_alter(
41    query: &str,
42) -> PolarsResult<Result<Statement, PartitionAlter>> {
43    Ok(match parse_statements_or_alter(query)? {
44        Err(alter) => Err(alter),
45        Ok(statements) => {
46            let [stmt]: [Statement; 1] = statements.try_into().map_err(|_| {
47                polars_err!(SQLInterface: "one (and only one) statement can be parsed at a time")
48            })?;
49            Ok(stmt)
50        },
51    })
52}
53
54/// The partition change at the start of `parser`; if there is none, `parser` stays at the start.
55fn parse_partition_alter(parser: &mut Parser) -> PolarsResult<Option<PartitionAlter>> {
56    // Up to the keywords that make this a partition change. Anything else is parsed as usual.
57    let head = parser
58        .maybe_parse(|p| {
59            // as sqlparser, which skips empty statements
60            while p.consume_token(&Token::SemiColon) {}
61            p.expect_keywords(&[Keyword::ALTER, Keyword::TABLE])?;
62            let if_exists = p.parse_keywords(&[Keyword::IF, Keyword::EXISTS]);
63            let only = p.parse_keyword(Keyword::ONLY);
64            let name = p.parse_object_name(false)?;
65            let reset = if p.parse_keywords(&[Keyword::SET, Keyword::PARTITIONED, Keyword::BY]) {
66                false
67            } else if p.parse_keywords(&[Keyword::RESET, Keyword::PARTITIONED, Keyword::BY]) {
68                true
69            } else {
70                return Err(ParserError::ParserError(String::new()));
71            };
72            Ok((name, if_exists, only, reset))
73        })
74        .map_err(to_sql_interface_err)?;
75    let Some((name, if_exists, only, reset)) = head else {
76        return Ok(None);
77    };
78    polars_ensure!(!only, SQLInterface: "`ALTER TABLE ONLY` is not supported");
79    let fields = if reset {
80        Vec::new()
81    } else {
82        parser
83            .expect_token(&Token::LParen)
84            .map_err(to_sql_interface_err)?;
85        polars_ensure!(
86            !parser.consume_token(&Token::RParen),
87            SQLInterface: "`SET PARTITIONED BY ()` needs a partition field; `RESET PARTITIONED BY` removes the partitioning"
88        );
89        let fields = parser
90            .parse_comma_separated(Parser::parse_expr)
91            .map_err(to_sql_interface_err)?;
92        parser
93            .expect_token(&Token::RParen)
94            .map_err(to_sql_interface_err)?;
95        fields
96    };
97    polars_ensure!(
98        !parser.consume_token(&Token::Comma),
99        SQLInterface: "`ALTER TABLE` takes one operation per statement"
100    );
101    let ended = parser.consume_token(&Token::SemiColon);
102    while parser.consume_token(&Token::SemiColon) {}
103    let next = parser.peek_token();
104    if next.token != Token::EOF {
105        polars_ensure!(
106            !ended,
107            SQLInterface: "one (and only one) statement can be parsed at a time"
108        );
109        return parser
110            .expected("end of statement", next)
111            .map_err(to_sql_interface_err);
112    }
113    Ok(Some(PartitionAlter {
114        name,
115        if_exists,
116        fields,
117    }))
118}
